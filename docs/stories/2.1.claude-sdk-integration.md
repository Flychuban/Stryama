### Client Singleton Pattern
```typescript
// lib/integrations/claude/client.ts
import Anthropic from '@anthropic-ai/sdk';
import { env } from '@/env.mjs';

let claudeClient: Anthropic | null = null;

export function getClaudeClient(): Anthropic {
  if (!claudeClient) {
    claudeClient = new Anthropic({
      apiKey: env.ANTHROPIC_API_KEY,
      timeout: env.ANTHROPIC_TIMEOUT_MS,
      maxRetries: 3,
    });
  }
  return claudeClient;
}

export async function healthCheck(): Promise<boolean> {
  try {
    const client = getClaudeClient();
    // Perform lightweight API call to verify connectivity
    // Implementation depends on SDK capabilities
    return true;
  } catch (error) {
    console.error('Claude API health check failed:', error);
    return false;
  }
}
```
